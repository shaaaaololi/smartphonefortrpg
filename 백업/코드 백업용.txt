               function deleteSelectedFolder() {
        let selectBox = document.getElementById("existingFolderSelect");
        let selectedFolder = selectBox.value;
        if (!selectedFolder || selectedFolder === "") {
          alert("삭제할 폴더를 선택해주세요!");
          return;
        }
        if (
          confirm(`정말로 [${selectedFolder}] 폴더를 영구 삭제하시겠습니까?`)
        ) {
          db.ref("trpg_sessions/" + selectedFolder)
            .remove()
            .then(() => {
              alert(`[${selectedFolder}] 폴더가 삭제되었습니다.`);
              loadFoldersIntoSelect();
            });
        }
      }

      function bootWithSelectedFolder() {
        let selectBox = document.getElementById("existingFolderSelect");
        let selectedFolder = selectBox.value;
        if (!selectedFolder || selectedFolder === "") {
          alert("접속할 폴더를 선택해주세요!");
          return;
        }
        startAppWithFolder(selectedFolder);
      }

      function bootWithNewFolder() {
        let newFolderInput = document
          .getElementById("newFolderNameInput")
          .value.trim();
        if (!newFolderInput) {
          alert("새로운 폴더명을 입력해주세요!");
          return;
        }
        startAppWithFolder(newFolderInput.replace(/[\.#$\[\]]/g, "_"));
      }

        let boot = document.getElementById("bootScreen");
        boot.style.opacity = "0";
        setTimeout(() => {
          boot.style.display = "none";
        }, 800);
        initSessionListeners();

      function sendMessage() {
        let input = document.getElementById("msgInput");
        let text = input.value.trim();
        if (!text) return;

        let selVal = document.getElementById("senderSelect").value;
        let sName = "나",
          sType = "me";
        if (selVal === "gm") {
          sName = "[GM 연출]";
          sType = "gm";
        } else if (selVal.startsWith("member_")) {
          sName = selVal.replace("member_", "");
          sType = "other";
        }

        // 현재 전송 시각 계산 (오전/오후 HH:mm)
        let now = new Date();
        let hours = now.getHours();
        let minutes = now.getMinutes();
        let ampm = hours >= 12 ? "오후" : "오전";
        hours = hours % 12;
        hours = hours ? hours : 12; // 0시일 경우 12시로 표기
        let timeFormatted = `${ampm} ${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}`;

        if (!chatHistories[currentChatName])
          chatHistories[currentChatName] = [];
        chatHistories[currentChatName].push({
          senderName: sName,
          text: text,
          type: sType,
          time: timeFormatted,
        });

        saveDataToFirebase();
        input.value = "";
      }

      let callTimer = null;
      let callSeconds = 0;

      function startCallTimer(callType, name, phone, avatar) {
        stopCallTimer();
        callSeconds = 0;
        activeCallInfo = { name, phone, avatar, type: callType };

        callTimer = setInterval(() => {
          callSeconds++;
          let min = String(Math.floor(callSeconds / 60)).padStart(2, "0");
          let sec = String(callSeconds % 60).padStart(2, "0");
          document.getElementById("callStatusText").innerText =
            `통화 중 (${min}:${sec})`;
        }, 1000);
      }

      function stopCallTimer() {
        if (callTimer) {
          clearInterval(callTimer);
          callTimer = null;
        }
      }
